cmake_minimum_required(VERSION 3.18)
project(AdamAI VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

find_package(PkgConfig REQUIRED)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c)
set(CURSES_NEED_WIDE TRUE)
find_package(Curses REQUIRED)
find_package(Threads REQUIRED)
find_library(NCURSESW_LIBRARY NAMES ncursesw REQUIRED)

add_library(adam_net STATIC src/stream.c src/ollama.c)
target_include_directories(adam_net PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(adam_net PUBLIC PkgConfig::CURL PkgConfig::JSONC)
target_compile_definitions(adam_net PUBLIC _XOPEN_SOURCE=700)

add_executable(adam src/main.c)
target_link_libraries(adam PRIVATE adam_net ${NCURSESW_LIBRARY} ${CURSES_LIBRARIES} Threads::Threads)
target_include_directories(adam PRIVATE ${CURSES_INCLUDE_DIRS})
target_compile_definitions(adam PRIVATE _XOPEN_SOURCE_EXTENDED=1)

if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(adam PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(adam_net PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(adam_stream_test tests/stream_test.c src/stream.c)
    target_include_directories(adam_stream_test PRIVATE src)
    target_link_libraries(adam_stream_test PRIVATE PkgConfig::JSONC)
    add_test(NAME ollama_stream_parser COMMAND adam_stream_test)
endif()
